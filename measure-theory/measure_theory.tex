\section{Lebesgue integral}

\subsection{Definition, basic properties and monotone convergence}
\begin{mydef} Let $X$ be a measurable space. We say that $f:X \to \bC$ is a simple function if it is measurable and the range $f(X) = \{f(x) ~|~ x \in X\}$ is a finite set. \end{mydef}

\begin{lemma} Let $f:X \to \bC$ be measurable. Then $f$ is a simple function if and only if $$f = \sum_{i=1}^n c_i \mathds{1}_{A_i}$$ for some positive integer $n$ and measurable $A_1, \ldots, A_n \subset X$ and $c_i \in \bC$.
\end{lemma}

\begin{proof} If $f$ is simple then the range of $f$ is a finite set $\{c_1, \ldots, c_n\}$. So we can just take $A_i = f^{-1}(c_i)$. 

Conversely, suppose $f$ takes such a form. Define now $q:X \to \{0,1\}^n$ by the map $$f(x) = (\mathds{1}_{A_1}(x), \ldots, \mathds{1}_{A_n}(x)).$$ Thus $f(x) = \langle q(x), c \rangle$ where $c = (c_1, \ldots, c_n)$. So clearly $f$ takes finitely many values (at most $2^n$), as it only depends on $q(x)$. \end{proof}

Thus, the space of simple functions form a vector space.

We say that $A_1, \ldots, A_n$ is a measurable partition of $X$ if each $A_i \subset X$ is measurable, $A_i \cap A_j = \emptyset$ for $i \neq j$ and $$\bigcup_{i=1}^n A_i = X.$$

\begin{lemma}\label{lemma: simple func int partition} Let $\mu$ be a measure on a space $X$ and suppose $f:X \to \bC$ is measurable. Suppose that $$f = \sum_{i=1}^n c_i\mathds{1}_{A_i},$$ for some positive integer $n$, some measurable partition $A_1, \ldots, A_n \subset X$ of $X$ and some $c_i \in \bR$. Then the quantity $$I_{\mu}(f) := \sum_{i=1}^n c_i \mu(A_i)$$ does not depend on $n$ and the $(A_1,c_1), \ldots, (A_n, c_n)$.

\end{lemma}

\begin{proof} Let $q_1, \ldots, q_m$ be all the possible distinct values of $c_i$, for $i \in \{1, \ldots, n\}$ such that $A_i \neq \emptyset$. Let $$I_j = \{ i \in \{1, \ldots, n\} ~|~ c_i = q_j \}.$$ Let $$B_j = \bigsqcup_{i \in I_j} A_i.$$ Note that the $B_j$ form a measurable partition of $X$ into non-empty subsets. Thus $$f = \sum_{j=1}^m q_j \sum_{i \in I_j} \mathds{1}_{A_i} = \sum_{i=1}^m q_j \mathds{1}_{B_j}. $$ Likewise, $$\sum_{i=1}^n c_i \mu(A_i) = \sum_{j=1}^m q_j \sum_{i \in I_j} \mu(A_i) = \sum_{j=1}^m q_j \mu(B_j).$$ If $x \in B_j$ then $f(x) = q_j$ and so as the $B_j$ are disjoint and non-empty, we thus have that $$f^{-1}(q_j) = B_j.$$ This shows that $$\sum_{i=1}^n c_i \mu(A_i) = \sum_{y \in \bR} y \mu(f^{-1}(y))$$ and is thus not dependent on the choice of $A_1, \ldots A_n$ nor the $c_1, \ldots, c_n$. \end{proof}

\begin{lemma}\label{lemma: I additive} Suppose that $f,g:X \to \bC$ are simple functions. Then $$I_{\mu}(f+g) = I_{\mu}(f) + I_{\mu}(g).$$

\end{lemma}

\begin{proof} We can write $$f = \sum_{i=1}^n c_i \mathds{1}_{A_i}$$ and $$g = \sum_{i=1}^m d_i \mathds{1}_{B_i}.$$ Now let $$Q_{i,j} = A_i \cap B_j.$$ Thus $$f = \sum_{i=1}^nc_i\mathds{1}_{A_i} = \sum_{i=1}^n c_i\sum_{j=1}^m \mathds{1}_{Q_{i,j}} = \sum_{i,j} c_i \mathds{1}_{Q_{i,j}}.$$ Likewise $$g = \sum_{i,j} d_j \mathds{1}_{Q_{i,j}}.$$ Thus by Lemma~\ref{lemma: simple func int partition} we have $$I_{\mu}(f+g) = \sum_{i,j} (c_i + d_j) \mu(Q_{i,j}) =  \sum_{i,j} c_i \mu(Q_{i,j}) + \sum_{i,j} d_j \mu(Q_{i,j}) = I_{\mu}(f) + I_{\mu}(g).   $$

\end{proof}

\begin{mydef} Let $\mu$ be a measure on a space $X$. Let $f:X \to [0, \infty]$ be a positive measurable function. Define the Lebesgue integral to be $$\int_{X} f d\mu = \sup \{ I_{\mu}(f) ~|~ g:X \to [0,\infty) \text{ is a simple function with } 0 \leq g \leq f. \}.$$

\end{mydef}

\begin{lemma} If $f_1,f_2:X \to [0,\infty]$ are such that $0 \leq f_1 \leq f_2$ then $$0 \leq \int_X f_1 d\mu \leq \int_X f_2 d\mu.$$

\end{lemma}

\begin{proof} Obvious. \end{proof}

\begin{lemma}\label{lemma: I is int} If $f:X \to [0,\infty)$ is simple then $I_{\mu}(f) = \int_X f d\mu$. \end{lemma}
\begin{proof} Clearly $I_{\mu}(f) \leq \int_X f d\mu$ since we can take $f=g$. Now we prove the other inequality. Thus suppose $g:X \to [0,\infty) $ is simple such that $0 \leq g \leq f$. Then $0 \leq f-g$ and $f-g$ is simple so $I_{\mu}(f-g) \geq 0$. However $$I_{\mu}(f) = I_{\mu}(g) + I_{\mu}(f-g) \geq I_{\mu}(g).$$ Thus $I_{\mu}(f) \geq \int_X f d\mu$. \end{proof}

\begin{thm}[Monotone convergence theorem] Suppose that $f_1,f_2,\ldots: X \to [0,\infty]$ be a sequence of measurable functions that is pointwise increasing, i.e., such that $f_{n}(x) \leq f_{n+1}(x)$ for all $x \in X$ and $n \in \bZ_{>0}$. Then $$\int_{X} f_i d\mu \to \int_{X} \lim_{n\to \infty} f_n(x).$$

\end{thm}

\begin{proof} Let $f(x) = \lim_{n \to \infty} f_n(x)$. Let $I = \int_X f_n d\mu$. If $I = 0$ then the theorem is true, so assume $I>0$. Hence let $0< I_1 < I_2 < \ldots <I$ be a sequence such that $I_{n} \to I$, i.e., if $I < \infty$ then we can let $I_n = I - n^{-1}$ otherwise we can let $I_n = n$. Fix an integer $Q>0$. Now there exists a simple function $g:X \to [0,\infty)$ such that $0\leq g \leq f$ and 
\begin{align}\label{eqn: g > I_Q} \int g d\mu > I_Q.\end{align}
Now write $$g = \sum_{i=1}^m c_i \mathds{1}_{A_i}$$ for some measurable disjoint $A_1, \ldots, A_m \subset X$ and $0 < c_i < \infty$. 
For convenience, let us consider two cases. 

Case 1: $\mu(A_i) = \infty$ for some $A_i$.

Then in this case $I = \infty$ as $\int_{X} g d\mu \geq c_i \mu(A_i) \infty$. Now let $$B_n = \{ x \in A_i ~|~ f_n(x) > c_i/2 \}.$$ Note that $$\bigcup_{n} B_n = A_i,$$ and thus $\mu(B_n) \to \mu(A_i) = \infty$. Now $f_n(x) \geq \frac{c_i}{2}\mathds{1}_{B_n}$ and hence $$\int_{X} f_n d\mu \geq \frac{c_i}{2}\mu(B_n) \to \infty,$$ which completes the proof of the monotone convergence theorem in this case.

Case 2: $\mu(A_i) < \infty$ for each $A_i$.

From (\ref{eqn: g > I_Q}), we can now fix $0 < \epsilon < 1$ such that $(1-\epsilon)\int_X g d > I_Q.$ Let 
$$B_{n,i} =  \{ x \in A_i ~|~ f_n(x) > (1-\epsilon)c_i \}.$$  Now 
$$\int_{X} f_n d\mu \geq \sum_{i=1}^m (1-\epsilon)c_i \mu(B_{i,n}) \to \sum_{i=1}^m (1-\epsilon)c_i \mu(A_i) \quad \text{as } n \to \infty.$$

But $$\sum_{i=1}^m (1-\epsilon)c_i \mu(A_i) = (1-\epsilon) \int_{X} g d\mu > I_Q. $$

Thus we have shown that for large enough $n$ we have $$\int_X f_n d\mu > I_Q,$$
which completes the proof. \end{proof}

\begin{prop} Let $f:X \to [0, \infty]$ be measurable. Then there exists measurable $A_1,A_2, \ldots \subset X$ and such that $$f(x) = \sum_{n=1}^{\infty} \frac{1}{n} \mathds{1}_{A_n}.$$ Thus $f$ is a monotonic limit of simple functions.

\end{prop}

\begin{proof} We define $A_1,A_2,\ldots$ recursively as follows. Let $$f_n = \sum_{i=1}^n \frac{1}{i} \mathds{1}_{A_i}.$$ We let $f_0 = 0$. Let $n \geq 0$ and suppose $0 \leq f_n \leq f$ has already been constructed (more precisely, we mean $A_1,\ldots, A_n$ have already been constructed). Let $$A_{n+1} = \{ x \in X ~|~ f_n(x) + \frac{1}{n+1} \leq f(x).\}$$ Thus $$f_{n+1} = f_n + \frac{1}{n+1} \mathds{1}_{A_{n+1}} \leq f$$ still holds.

We must now show that $$f_n \to f.$$ First, if $f(x) = \infty$ then $x \in \mathds{1}_{A_n}$ for all $n$ and thus $$\sum_{n=1}^{\infty} \frac{1}{n}\mathds{1}_{A_n}(x) = \sum_{n=1}^{\infty} \frac{1}{n} = \infty = f(x).$$
Now suppose that $f(x) < \infty$. Suppose for contradiction that $$\lim_{k \to \infty} f_k(x) < f(x) - 1/N$$ for some $N>0$. As $f_n$ is increasing, it now follows that for $n>N$ we have $$f_n(x) \leq \lim_{k \to \infty} f_k(x) \leq f(x) - \frac{1}{N} \leq f(x) - \frac{1}{n+1}.$$ But this precisely means that $x \in A_n$ for all $n>N$. Thus $$f(x) \geq \sum_{n=N+1}^{\infty}\frac{1}{n} = \infty,$$ which is a contradiction. \end{proof}

\subsection{Additivity of the integral}
\begin{thm}\label{thm: int additive for positive} Suppose that $f,g:X \to [0,\infty]$ are measurable. Then $$\int_X (f+g)d\mu = \int_X f d\mu + \int_X g d\mu.$$
\end{thm}

\begin{proof} We know this is true for $f,g$ simple by Lemma~\ref{lemma: I additive} and Lemma~\ref{lemma: I is int}. Now we can write $f$ and $g$ as monotonic limits of simple functions $f_n$, $g_n$. But $f_n + g_n \to f+g$ monotonically. So the monotone convergence theorem completes the proof. \end{proof}

\begin{mydef} We let $\int_A f d\mu = \int_X \mathds{1}_A f d\mu$. We now define $$\int_{X} f d\mu = \int_{f \geq 0} f d\mu - \int_{f < 0} |f| d\mu,$$ in case it is not the case that these integrals are not both $\infty$ (in which case, it is undefined).
\end{mydef}

Observe that in the definition above, we can replace strict inequality with non-strict inequality and vice versa (this is because $\int_X 0 d\mu = 0$ and the integral is additive on non-negative functions.
Also, let us define the positive part $f^+ = \mathds{1}_{f \geq 0} f$ and negative part $f^- = -\mathds{1}_{f < 0}f$ of $f$. Thus by definition we have $f^+$ and $f^-$ are both $\geq 0$ and $$\int_X f d\mu = \int_X f^+ d\mu - \int_X f^- d\mu$$ under the assumption that at least one of these integrals is not infinite.

\begin{lemma} Suppose that $f,g:X \to [0,\infty]$ are measurable with $\int_X g d\mu < \infty$ or $\int_X f d\mu < \infty$. Then $$\int_X (f-g) d\mu = \int_X f d\mu - \int_X g d\mu$$ with the left hand side being well defined (either the positive or negative part has finite integral). \end{lemma}

\begin{proof} By definition 
\begin{align*} \int_X (f-g) d\mu &= \int_X (f-g)^+ d\mu - \int_X (g-f)^+ d\mu \\
&= \int_{f \geq g} (f-g) - \int_{g > f} (g-f) d\mu
\end{align*} 
but $(f-g)^+ \leq f$ and $(g-f)^+ \leq g$ so at least one of these integrals on the right hand side is non-infinite. Thus the integral is well defined.
Now by Theorem~\ref{thm: int additive for positive}, we have 
\begin{align} \label{eqn f-g int} \int_{f \geq g} (f-g)d\mu + \int_{f \geq g} g d\mu = \int_{f \geq g} f d\mu \end{align}
and 
\begin{align} \label{eqn g-f int} \int_{g > f} (g-f) d\mu + \int_{g>f} f d\mu = \int_{g>f} g d\mu.  \end{align}
We will now assume $\int_X f d\mu < \infty$ (the other case will then follow by symmetry).
Thus the right hand side of (\ref{eqn f-g int}) is non-infinite and so all terms are non-infinite. So we can safely rearrange this equation. Likewise, the term $\int_{g>f} d \mu <\infty$ so it can safely be moved to the right hand side in the other equation (\ref{eqn g-f int}). After rerranging and subtracting these two equations, we obtain that 

$$\int_X (f-g) d\mu = \int_{f \geq g} f d\mu - \int_{f \geq g} g d\mu - \left(\int_{g>f} g d\mu -\int_{g>f} f d\mu\right).  $$

Only the third integral can possibly be infinite as argued above, thus the terms can be safely grouped to give $$\int_X (f-g) d\mu = \int_{f\geq g} f + \int_{g>f} d \mu - \left(\int_{g>f} g d\mu + \int_{f \geq g} g d\mu\right).$$ Finally, the proof is complete using Theorem~\ref{thm: int additive for positive}. \end{proof}

\begin{thm} Suppose that $f,g:X \to (-\infty, \infty]$ are such that either $$\int_{f \geq 0} f d\mu + \int_{g \geq 0} g d\mu < \infty$$ or $$\int_{f < 0} f d\mu + \int_{g < 0} g d\mu > -\infty.$$

Then the following holds and is well defined (no $\infty - \infty$ terms and all integrals exist) $$\int_X (f+g) d\mu = \int_X f d\mu + \int_X g d\mu.$$

\end{thm}

\begin{proof} Write $f = f^+ - f^-$ and $g = g^+ - g^-$ as above. We assume the first condition, thus $$\int_X f^+ + g^+ d\mu < \infty.$$ Now $$f+g = f^+ + g^+ - (f^- + g^-),$$ so the assumption of the previous theorem must apply (likewise, they also apply for the other case). After applying this theorem, rearranging and using additivity of the integral for positive functions the proof is complete. \end{proof}

\subsection{Integrable functions and convergence}

We say that a measurable $f:X \to \bC$ is integrable if $\int_X |f| d\mu <\infty$.

\begin{lemma} Suppose that $f:X \to \bC$ is integrable. Then for $\epsilon>0$ there exists a measurable $A \subset X$ such that $\mu(A) < \infty$ and $$\int_{X \setminus A} |f| d\mu < \epsilon.$$\end{lemma}

\begin{proof} Let $0 \leq h_1 \leq h_2 \leq \ldots$ be a sequence of simple functions converging monotinically to $|f|$. Thus $\int_{X} h_n d\mu < \infty$ and so, as $h_n$ is simple, there is a measurable set $A_n$ such that $\mu(A_n) < \infty$ and $h_n(x) = 0$ for $x \in X \setminus A$. By the monotone convergence theorem, there is a large enough $n$ so that $$\int_X (|f| - h_n) d\mu < \epsilon.$$ However $$\epsilon > \int_{X}(|f|-h_n) d\mu = \int_{A_n}(|f| - h_n) + \int_{X \setminus A_n} |f| d\mu$$ as $h = 0$ on $X \setminus A_n$. So we can take $A = A_n$ as all integrands are non-negative. \end{proof}

\begin{thm}[Lebesgue dominated convergence theorem] Suppose that $f_1,f_2,\ldots :X \to \bC$ are measurable and $g:X \to [0,\infty]$ is integrable, i.e, $\int_X g d\mu < \infty$ such that $|f_n| \leq g$ for all $f_n$. Suppose that $f_n \to f$ pointwise, then $f$ is integrable and $$\int_X |f_n - f| d\mu \to 0,$$ 
thus
$$\int_X f_n d\mu \to \int_X f d\mu.$$

\end{thm}

\begin{proof} Note that $|f| \leq |g|$ thus $|f|$ is integrable. Fix $\epsilon>0$ and use the previous lemma to find $A \subset X$ with $\mu(A)<\infty$ and $\int_{X \setminus A} |g|d\mu < \epsilon$. Thus $$\int_{X \setminus A} |f - f_n| \leq \int_{X \setminus A} 2g d\mu < 2\epsilon.$$ Now as $\mu(A)<\infty$, we can choose $\delta >0$ so that $$\delta \mu(A) < \epsilon.$$ 

Let $$A_n = \{ x \in A ~|~ |f_m(x) - f(x)| < \delta \text{ for all } m \geq n. \}.$$

Notice that $A_n \subset A_{n+1}$ and $\bigcup_n A_n = A$. Thus by the monotone convergence theorem, we have $$|\int_{A_n} g - \int_{A} g| < \epsilon$$ for all large enough $n > N$. This means that $$\int_{A \setminus A_n} g < \epsilon$$ for all $n>N$ and consequently $$\int_{A \setminus A_n} |f - f_n| \leq \int_{A \setminus A_n} 2g < 2\epsilon .$$ Finally, $|f(x) - f_n(x)| < \delta$ for all $x \in A_n$, by definition of $A_n$. Thus $$\int_{A_n} |f(x) - f_n(x)| \leq \mu(A) \delta <\epsilon.$$ By splitting up $\int_X = \int_{X\setminus A} + \int_{A \setminus A_n} + \int_{A_n}$ and combining the estimates gathered, we obtain $$\int_X |f - f_n| d\mu < 2\epsilon + 2\epsilon + \epsilon = 5 \epsilon$$ for $n>N$. Thus completing the proof. \end{proof}

\section{Outer measures}

\begin{mydef} An \textit{outer measure} on a set $X$ is a function $\mu:2^X \to [0,\infty]$ which satisfies the following conditions:

\begin{enumerate}
	\item Countably sub-additivity: if $A_1,A_2,\ldots \subset X$ then $$\mu(\bigcup_{i=1}^{\infty} A_i) \leq \sum_{i=1}^{\infty} \mu(A_i).$$
	\item Monotonicity: If $A \subset B \subset X$ then $\mu(A) \leq \mu(B)$.
	\item Empty set is null: $\mu(\emptyset) = 0$.
\end{enumerate}

We say that $A \subset X$ is $\mu$-measurable if $$\mu(B) = \mu(B \cap A) + \mu(B \cap A^c) \quad \text{for all } B \subset X.$$ Note that, by sub-additivity, this is equivalent to $$\mu(B) \geq \mu(B \cap A) + \mu(B \cap A^c) \quad \text{for all } B \subset X.$$

\end{mydef}

\begin{thm}\label{thm: mes sets sigma algebra} The set of all measurable subsets forms a $\sigma$-algebra on $X$. Moreover, $\mu$ is countably-additive (a measure) on $X$ for this $\sigma$-algebra.

\end{thm}

\begin{proof} Observe that $\emptyset$ is measurable since for $B \subset X$ we have $$\mu(B) = 0 + \mu(B \cap X) = \mu(\emptyset) + \mu(B \cap X) = \mu(B \cap \emptyset) + \mu(B \cap \emptyset^c).$$ Also $A$ is measurable then $A^c$ also is by symmetry.

We first show that the set of measurable sets is closed under finite intersection. Thus suppose that $A_1, A_2,\ldots \subset X$ are measurable and $B \subset X$.

As $A_1$ is measurable, we obtain that $$\mu(B \cap A_2^c) = \mu(B \cap A_2^c \cap A_1) + \mu(B \cap A_2^c \cap A_1^c)$$ and $$\mu(B \cap A_2) = \mu(B \cap A_2 \cap A_1) + \mu(B \cap A_2 \cap A_1^c).$$ 

Now we add these two equations and use the inequality (which follows from subaditivity and the identity $A_1 \cup A_2 = (A_1^c \cap A_2) \cup (A_1 \cap A_2^c) \cup (A_1 \cap A_2)$) $$\mu(B \cap A_2^c \cap A_1) + \mu(B \cap A_2 \cap A_1) + \mu(B \cap A_2 \cap A_1^c) \geq \mu(B \cap (A_1 \cup A_2))$$ to get that $$\mu(B \cap A_2^c) + \mu(B \cap A_2) \geq \mu(B \cap (A_1 \cap A_2)) + \mu(B \cap A_1^c \cap A_2^c).$$ Measurability of $A_2$ means that the left hand side is $\mu(B)$, thus $$\mu(B) \geq \mu(B \cap (A_1 \cap A_2)) + \mu(B \cap (A_1 \cup A_2)).$$ The reverse inequality holds from sub-additivity, thus $A_1 \cup A_2$ is measurable.

Thus we have shown that the measurable sets form an algebra. To show that they form a $\sigma$-algebra, it thus remains to show that a \textit{disjoint} union of measurable sets is measurable. Thus suppose $A_1, A_2, \ldots$ are disjoint measurable sets and $B \subset X$. We claim that $$\mu\left(B \cap \left(\bigcup_{i=1}^n A_i\right)\right) = \mu(B \cap A_1) + \cdots + \mu(B \cap A_n).$$ We provie this by induction as follows. Suppose it holds for $n$. Then by measurability of $A_{n+1}$ we have that 
\begin{align*}\mu\left(B \cap \left(\bigcup_{i=1}^{n+1} A_i\right)\right) &=  \mu\left(B \cap \left(\bigcup_{i=1}^{n+1} A_i\right) \cap A_{n+1}\right) + \mu\left(B \cap \left(\bigcup_{i=1}^{n+1} A_i\right) \cap A_{n+1}^c\right) 
\\ &= \mu(B \cap A_{n+1}) + \mu\left(B \cap \left(\bigcup_{i=1}^n A_i\right)\right) \end{align*}

which completes the induction step (the last equation uses disjointness of the $A_i$). Now by montonicity of $\mu$ we get $$\mu\left(B \cap \left(\bigcup_{i=1}^{\infty} A_i\right)\right) \geq \lim_{n \to \infty} \mu\left(B \cap \left(\bigcup_{i=1}^n A_i\right)\right) = \sum_{i=1}^{\infty} \mu(B \cap A_i).$$ The reverse inequality holds from sub-additivity, thus we have shown that $$\mu\left(B \cap \left(\bigcup_{i=1}^{\infty} A_i\right)\right) = \sum_{i=1}^{\infty} \mu(B \cap A_i).$$ In particular, for $B = X$ this shows that $\mu$ is countably-additive on the measurable sets. It remains to show that this countable union is measurable. Observe that since finite unions are measurable we have
\begin{align*} \mu(B) &= \mu\left(B \cap \left(\bigcup_{i=1}^n A_i\right)\right) + \mu\left(B \cap \left(\bigcup_{i=1}^n A_i\right)^c\right) \\
&\geq  \mu\left(B \cap \left(\bigcup_{i=1}^n A_i\right)\right) + \mu\left(B \cap \left(\bigcup_{i=1}^{\infty} A_i\right)^c\right). \end{align*} Now letting $n \to \infty$ and using countable sub-additivity, the proof is complete. \end{proof}


\begin{eg}[Lebesgue outer measure] If $I \subset \bR$ is a non-empty bounded interval, we let $len(I) = \sup(I) - \inf(I)$ and $len(\emptyset) = 0$. Note $I$ could be open, closed or half open. If $\cI$ is a countable set of bounded intervals of $\bR$, then we denote $len(\cI) = \sum_{I \in \cI} len(I)$. We say that $\cI$ covers $B$ if $$ B \subset \bigcup_{I \in \cI} I.$$ Finally, denote the Lebesgue outer-measure of $B \subset \bR$ to be $$\lambda(B) = \inf_{\cI \text{ covers } B} len(\cI).$$

Let us now show that any bounded interval $J$ is $\lambda$-measurable. To see this, let $B \subset \bR$ be any subset and suppose that $\cI$ covers $B$ such that $len(\cI) < \lambda(B) + \epsilon$. We can write $J^c = J_1 \cup J_2$ where $J_1$ and $J_2$ are disjoint (unbounded) intervals. Let $\cI_1 = \{I \cap J ~|~ I \in \cI\}$ and $$\cI_2 = \{I \cap J_m ~|~ I \in \cI, m \in \{1,2\}\}.$$ Then $\cI_1$ covers $B \cap J$ and $\cI_2$ covers $B \cap J^c$. Observe that $len(\cI_1) + len(\cI_2) = len(\cI)$ since $$len(I) = len(I \cap J_1) + len(I \cap J) + len(I \cap J_2).$$ Thus $$\lambda(B \cap J) + \lambda(B \cap J^c) \leq len(\cI_1) + len(\cI_2) =len(\cI) \leq \lambda(B) + \epsilon.$$ Letting $\epsilon \to 0$ shows that $J$ is $\lambda$-measurable.

\end{eg}

\section{Monotone classes}

\begin{mydef} Let $X$ be a set. A monotone class on $X$ is a collection of subsets $\cA \subset 2^X$ which satisfies the following properties.

\begin{enumerate}
	\item $\emptyset, X \in \cA$.
	\item If $A_1,A_2,\ldots \in \cA$ is such that $A_1 \subset A_2 \subset \ldots$ then $\bigcup_{i=1}^{\infty} A_i \in \cA$.
	\item If $A_1,A_2,\ldots \in \cA$ is such that $A_1 \supset A_2 \supset \ldots$ then $\bigcap_{i=1}^{\infty} A_i \in \cA$.
\end{enumerate}

\end{mydef}
 
Recall than an algebra on $X$ is a collection of subsets of $X$ that is closed under finite intersections, finite unions, complements and contains $\emptyset$ and $X$.

\begin{thm}[Monotone Class Lemma] Let $X$ be a set and suppose that $\cA$ is an algebra of subsets of $X$. Then the $\sigma$-algebra generated by $\cA$ is the smallest monotone class that contains $\cA$.

\end{thm}

\begin{proof} Let $\cM$ denote the smallest monotone class that contains $\cA$. We must show that $\cM$ is a $\sigma$-algebra. It is enough to show that $\cM$ is closed under finite unions and complements. For $A \in \cA$, let $$\cB_A = \{B \in \cM ~|~ B \cup A \in \cM\}.$$ Note that $\cA \subset \cB_A$ by definition of algebra. 

We claim that $\cB_A$ is a monotone class. To see this, suppose that $B_1 \subset B_2 \subset \ldots$ are all in $\cB_A$. Thus $B_i \cup A \in \cM$. But $B_i \cup A$ is a nested sequence of sets, and so $\bigcup_i B_i \cup A \in \cM$ and so $\bigcup_i B_i \in \cM$. Likewise suppose that $B_1 \supset B_2 \supset \ldots$ are all in $\cB_A$. Thus $B_i \cup A \in \cM$. As $B_i \cup A$ monotonically decreases to $A \cup \bigcap_i B_i$ we have that $A \cup \bigcap_i B_i \in \cM$ and so $\bigcap_i B_i \in \cB_A$.

Thus $\cB_A$ is a monotone class containing $\cM$, but is a subset of $\cM$ by definition. Thus $\cM = \cB_A$. Now let $$\cM_0 = \{ M_0 \in \cM ~|~ M_0 \cup M \in \cM \text{ for all } M \in \cM \}.$$ We see that $\cA \subset \cM_0$ since $\cM = \cB_A$. Note also that $\cM_0$ is a monotone class. Thus $\cM_0 = \cM$. Thus we have shown that $\cM$ is closed under finite unions, and hence countable unions.

Now let $\cM' = \{ M \in \cM ~|~ M^c \in \cM\}$. Note that $\cA \subset \cM'$ and $\cM'$ is a monotone class. Thus $\cM' = \cM$ and so $\cM$ is closed under complements. Thus $\cM$ is indeed a $\sigma$-algebra.

\end{proof}

\begin{eg} Let $\cA$ denote the set of all finite unions of half-open intervals $ \bigcup_{i=1}^n [a_i, b_i)$ for some $n \geq 0$. Then $\cA$ is an algebra, but not a $\sigma$-algebra. The monotone class lemma says that the smallest monotone class containing $\cA$ is the Borel $\sigma$-algebra.

Now let $\cU$ be the smallest set containing $\cA$ that is closed under countable nested unions. We claim that each element in $\cU$ is either empty or has positive Lebesgue measure. Let $\cU'$ denote all $U \in \cU$ with this property ($U$ is either empty or positive measure). Clearly $\cA \subset \cU'$. Now let $U_1,U_2,\ldots \in \cU'$ and $U = \bigcup_{i}U_i$. If $U$ has zero measure, then each $U_i$ has zero measure and so each $U_i$ empty and so $U$ is empty. Thus $U \in \cU'$. Thus $\cU = \cU'$. This means that $\cU$ is not the Borel $\sigma$-algebra. This shows that in the definition of monotone class, we really do need closure under countable intersections as well otherwise the monotone class lemma is not true.

\end{eg}

\section{Product measures and Fubini-Tonelli theorem}

Let $X,Y$ be measure spaces and let $$\cR(X,Y) = \{ U \times V ~|~ U \subset X \text{ measurable}, V \subset Y \text{measureable}\}$$ Denote the set of all rectangles with in $X \times Y$ with measurable projections onto the axes. Let $\mu$ and $\nu$ be two positive measures on $X$ and $Y$ respectively. Then we define the area $$Area(U \times V) = \mu(U)\nu(V).$$ If $\cC \subset \cR$ is a countable collections of rectangles define the $$Area(\cC) = \sum_{C \in \cC} Area(C).$$ We say that $\cC$ covers $A \subset X \times Y$ if $$A \subset \bigcup_{C \in \cC} C.$$ Define the product outer-measure $$( \mu \times \nu) (A) = \inf_{\cC \text{ covers } A} Area(\cC).$$ 
\begin{prop} Each element of $\cR$ is $\mu \times \nu$-measurable.

\end{prop}

\begin{proof} Let $R \in \cR$ and let $B \subset X \times Y$. Let $\cC$ be a countable collection of rectangles that covers $B$ such that $$Area(\cC) \leq (\mu \times \nu) (B) + \epsilon.$$ Observe that the complement $R^c$ can be written as a disjoint union of finitely many rectangles. More precisely, if $R = U \times V$ then $$R^c = U^c \times V^c \cup U^c \times V \cup U \times V^c.$$ Call these three rectangles $R_1,R_2,R_3$. Observe that any rectangle $C = C_1 \times C_2 \in \cC$ we have that
\begin{align*}&~~ Area(C \cap R) + Area(C \cap R_1) + Area(C \cap R_2) + Area(C \cap R_3) \\
&= \mu(C_1 \cap U) \nu(C_2 \cap V) + \mu(C_1 \cap U^c)\nu(C_2 \cap V^c) + \mu(C_1 \cap U^c)\nu(C_2 \cap V) + \mu(C_1 \cap U)\nu(C_2 \cap V^c) \\
&= (\mu(C_1 \cap U) + \mu(C_1 \cap U^c))(\nu(C_2 \cap V) + \nu(C_2 \cap V^c)) \\
&= \mu(C_1)\nu(C_1) \\
&= Area(C). \end{align*} 

Thus if we let $$\cC_2 = \{ C \cap R_i ~|~ i=1,2,3 \text{ and } C \in \cC \}$$ and $$\cC_1 = \{ C \cap R ~|~ C \in \cC\}$$ then $$Area(\cC_1) + Area(\cC_2) = Area(\cC).$$ Observe that $\cC_1$ covers $B \cap R$ and $\cC_2$ covers $B \cap R^c$ and so $$(\mu \times \nu)(R \cap B) + (\mu \times \nu)(R^c \cap B) \leq Area(\cC_1) + Area(\cC_2) = Area(\cC) \leq (\mu \times \nu) (B) + \epsilon.$$ By letting $\epsilon \to 0$, the proof is complete. \end{proof}

We let $\cB_{X \times Y}$ denote the smallest $\sigma$-algebra on $X \times Y$ containing $\cR$. Note that this Proposition and Theorem~\ref{thm: mes sets sigma algebra} says that $\mu \times \nu$ is a measure (countably sub-additive) on this $\sigma$-algebra.

Another convenient definition of $\mu \times \nu$ is as follows. Let $\cC$ is a countable collection of rectangles, then define the positive function $$F_{\cC} :X\times Y \to [0,\infty]$$ to be $$F_{\cC}(a) = \sum_{C \in \cC} \mathds{1}_{C}(a).$$ Then notice that $\cC$ covers $A$ if and only if $\mathds{1}_{A} \leq F_{\cC}$. Now consider the function $X \to [0,\infty]$
given by $$x \mapsto \int_{Y} F_{\cC}(x,y) d\nu(y).$$ Let us show that this function is well defined and measurable. If $F_{\cC} = \sum_{i=1}^{\infty} \mathds{1}_{U_i \times V_i}$ then $$\int_{Y} F_{\cC}(x,y) d\nu(y) = \int_{Y} \mathds{1}_{U_i}(x)\mathds{1}_{V_i}(y) d \nu(y) = \sum_{i=1}^{\infty}\mathds{1}_{U_i}(x) \nu(V_i)$$ by the monotone convergence theorem. Thus this function is indeed measurable (by being the countable sum of non-negative measurable functions). By applying the monotone convergence theorem again we have the useful Fubini-type equation $$\int_X \int_Y F_{\cC}(x,y) d\nu(y) d\mu(x) = \sum_{i=1}^{\infty} \mu(U_i)\nu(V_i) = Area(\cC).$$

\begin{prop} If $R \in \cR$ is a rectangle, then $(\mu \times \nu)(R) = Area(R)$.

\end{prop}

\begin{proof} Suppose that $\cC$ covers $R$. Then we must have that $$F_{\{R\}} = \mathds{1}_{R} \leq F_{\cC}.$$ Thus we must have that $$Area(R) = \int_{X}\int_{Y}F_{\{R\}} (x,y) d\nu(y)d\mu(x) \leq \int_{X}\int_{Y} F_{\cC}(x,y) d\nu(y)d\mu(x) = Area(\cC).$$ As $\cC$ was arbitrary, this shows that $$(\mu \times \nu) (R) \geq Area(R).$$ The reverse inequality holds because $\{R\}$ covers $R$. \end{proof}

\begin{thm}[Fubini-Tonelli theorem for indicator functions] Suppose that $\mu$ and $\nu$ are $\sigma$-finite (on $X$ and $Y$ respectively). Let $B \in \cB_{X \times Y}$. Then the mapping $$x \mapsto \int_{Y} \mathds{1}_B(x,y) d\nu(y)$$ is measurable and its integral over $X$ is $(\mu \times \nu)(B)$, that is, $$\int_{X} \int_{Y} \mathds{1}_B(x,y) d\nu(y) d\mu(x) = (\mu \times \nu)(B) = \int_{Y} \int_{X} \mathds{1}_B(x,y) d\mu(x) d\nu(y) .$$

\end{thm}

\begin{proof} By a partitioning argument and the monotone convergence theorem, we can reduce to the case where $\mu$ and $\nu$ are both finite.

Let $\cM$ denote the set of all $B \in \cB_{X \times Y}$ for which this theorem is true. By the proposition above, $\cR \subset \cM$. Note that a finite union of rectangles may be written as a finite union of disjoint rectangles, thus the smallest algebra that contains $\cR$ is a subset of $\cM$. Thus, by the montone class lemma it will be enough to show that $\cM$ is a montone class. That $\cM$ is closed under countable nested unions follows from the monotone convergence theorem. To show that $\cM$ is closed under countable nested intersections, we can use the fact that $\cM$ is closed under complements, which uses the finiteness of $\mu$ and $\nu$ as follows. Indeed if $B \in \cM$, then $$\int_{X} \int_{Y} \mathds{1}_{B^c}(x,y) d\nu(y) d\mu(x) = \int_{X} \int_{Y} (1-\mathds{1}_B(x,y)) d\nu(y) d\mu(x) = (\mu \times \nu)(X \times Y) - (\mu \times \nu)(B) = (\mu \times \nu)(B^c)$$
and so $B^c \in \cM$. \end{proof}

\begin{eg} The $\sigma$-finiteness assumption is necessary. For instance, let $X,Y = \bR$ and let $\mu$ be the counting measure (not $\sigma$-finite) and let $\nu$ be the Lebesgue measure and let $B = \{(x,x) ~|~ x \in \bR\}$ be the diagonal. Then $$\int_{X} \int_{Y} \mathds{1}_B(x,y) d\nu(y) d\mu(x)  = \int_X 0 d\mu(x) = 0$$ and yet  $$\int_{Y} \int_{X} \mathds{1}_B(x,y) d\mu(x) d\nu(y) = \int_Y 1 d\nu(y) = \infty. $$

\end{eg}

By using the monotone convergence theorem, we get the Fubini-Tonelli theorem for positive measurable functions.

\begin{thm}[Tonelli theorem]  Suppose that $\mu$ and $\nu$ are $\sigma$-finite (on $X$ and $Y$ respectively) and suppose that $f:X \times Y \to [0, \infty]$ is $\cB_{X \times Y}$-measurable. Then $$x \mapsto \int_{Y} f(x,y) d\nu(y)$$ is measurable and
$$\int_{X} \int_{Y} f(x,y) d\nu(y) d\mu(x)  = \int_{X \times Y} f(x,y) d(\mu \times \nu)(x,y)  = \int_{Y} \int_{X} f(x,y) d\mu(x) d\nu(y). $$
\end{thm}

\begin{proof} Write $f = \sum_{i=1}^{\infty} c_i\mathds{1}_{B_i}$ for some $B_i \in \cB_{X \times Y}$ and $c_i \in [0,\infty)$. Now apply the previous theorem and the monotone convergence theorem. \end{proof}


\begin{thm}  Suppose that $\mu$ and $\nu$ are $\sigma$-finite (on $X$ and $Y$ respectively) and suppose that $f:X \times Y \to [-\infty, \infty]$ is $\cB_{X \times Y}$-measurable. Suppose that\footnote{By Tonelli's theorem, we can replace this condition with an iterated integral for $|f|$.} $$\int_{X \times Y} |f(x,y)| d(\mu \times \nu)(x,y) < \infty.$$

Then there exists a measurable set $X' \subset X$ that is $\mu$-conull and such that $$x \mapsto \int_{Y} |f(x,y)| d\nu(y) < \infty \quad\text{for all } x \in X'$$ and the map $X' \to X$ given by $$x \mapsto \int_{Y} f(x,y) d\nu(y) $$ is a well defined measurable map. Furthermore\footnote{The integrands are only almost everywhere well defined. For example the first integral is over $X'$ rather than $X$, but as $X'$ is co-null we can still sensibly integrate over $X$ by extending the domain to whichever function on $X$ we like.} $$\int_{X} \int_{Y} f(x,y) d\nu(y) d\mu(x)  = \int_{X \times Y} f(x,y) d(\mu \times \nu)(x,y)  = \int_{Y} \int_{X} f(x,y) d\mu(x) d\nu(y). $$

\end{thm}

\begin{proof} As $f$ is $L^1$ with respect to $\mu \times \nu$, we can write $f = f^+ - f^-$ where $f^+, f^-:X \times Y \to [0,\infty]$ and $f^+, f^-$ are both $L^1$. Thus by an application of Tonelli's theorem to both of these functions we have that $$\int_X \int_Y f^{\pm}(x,y) d\nu(y) d\mu(x) < \infty$$ and so there must be a co-null set $X' \subset X$ so that $$\int_Y f^{\pm}(x,y) d\nu(y) < \infty \quad \text{for all } x \in X'$$ and this quantity is a measurable function on $X'$.  Thus 
\begin{align*}\int_{X'} \int_Y f (x,y) d\nu(y) d\mu(x) &= \int_{X'} \int_Y \left(f^+(x,y) - f^-(x,y)\right) d\nu(y) d\mu(x) \\
&= \int_{X'}\left(\int_Y f^+(x,y) d\nu(y) - \int_Y f^-(x,y) d\nu(y) \right) d\mu(x) \\
&=  \int_{X'}\int_Y f^+(x,y) d\nu(y)d\mu(x) - \int_{X'}\int_Y f^-(x,y) d\nu(y)d\mu(x) \\ 
&=  \int_{X}\int_Y f^+(x,y) d\nu(y)d\mu(x) - \int_{X}\int_Y f^-(x,y) d\nu(y)d\mu(x) \\ 
&=  \int_{X \times Y} f^+(x,y) d(\mu \times \nu)(x,y) - \int_{X \times Y} f^-(x,y) d(\mu \times \nu)(x,y) \\
&= \int_{X \times Y} f(x,y) d(\mu \times \nu)(x,y)  \end{align*}
\end{proof}

\section{Jensen's Inequality}

\begin{mydef} We say $U \subset \bR^d$ is convex if $$tu + (1-t)v \in U$$ whenever $u,v \in U$ and $t \in [0,1]$. We say that $\varphi:U \to \bR$ is convex if $$\varphi(tu + (1-t)v) \leq t \varphi(u) + (1-t)\varphi(v)$$ for all $u,v \in U$ and $t \in [0,1]$. Note that this is equivalent to saying that the region $$\{(u,y) \in U \times \bR ~|~ y \geq \varphi(u) \}$$ is convex.

\end{mydef}

\begin{prop} Let $U \subset \bR^d$ be an open convex set and suppose that $\varphi:U \to \bR$ is convex. Then $\varphi$ is continuous.

\end{prop}

\begin{remark} We really need the assumption that $U$ is open. For example, if $U = [0,\infty)$ then the function $\varphi(u) = \mathds{1}_{\{0\}}(u)$ which vanishes every but equals $1$ for $u=0$ is convex but not continuous.

\end{remark}

\begin{proof} We prove this by induction on $d$. The base case $d=0$ is trivial (where $U = \{0 \} = \bR^{0}$ is a zero-dimensional vector space). Now fix $x_0 \in U$. Let $U' \subset U$ be a cube contained in $U$ such that $x_0$ is the center of $U'$. Let $B$ be the boundary of this cube $U$. Observe that by the induction hypothesis, $\varphi$ is continuous on $B$ (as $B$ is made up of finitely many $d-1$-dimensional faces) and thus is bounded on $B$, say $|\varphi(b)| \leq M$ for all $b \in B$. Observe also that any $u \in U'$ may be written as $u = tx_0 + (1-t)b$ for some $b \in B$ (to see this, note that any ray starting at $x_0$ that goes through $u$ must intersect $B$ at some point $b$). 
Now observe that $$\|u - x_0\| = \|(t-1)x_0 + (1-t)b\| = (1-t)\|b - x_0\| \geq (1-t)R$$ where $R>0$ is half the side-length of the box $U'$. 
Now by convexity we have that 
\begin{align*} \varphi(u) &= \varphi(tx_0 + (1-t)b) \leq t\varphi(x_0) + (1-t)\varphi(b) \\
&\leq t \varphi(x_0) + (1-t)M \\ 
&\leq t\varphi(x_0) + MR^{-1}\|u - x_0\| \\
&= \varphi(x_0) + (t-1)\varphi(x_0) + MR^{-1} \|u - x_0\| \\
&\leq \varphi(x_0) + R^{-1}(|\varphi(x_0)| + M)\|u-x_0\|. \end{align*}
 Thus $$\limsup_{u \to x_0} \varphi(u) \leq \varphi(x_0).$$ Now suppose for contradiction, that there is some $u_1,u_2, \ldots \in U$ such that $u_i \to x_0$ but $\varphi(u_i) < \varphi(x_0) - \epsilon$ for some fixed $\epsilon > 0$.
 Now observe that $$x_0 = t_iu_i + (1-t_i)b_i$$ for some $t_i \in [0,1]$ and $b_i \in B$ as we can see by considering a ray that starts at $u_i$ and passes through $x_0$ and then letting $b_i \in B$ be a point on this ray.
We see that $t_i \to 1$ since $$\|x_0 - u_i\| = \|(t_i-1)u_i + (1-t_i)b_i \| = (1-t_i)\|b_i - u_i\|$$ and the left hand side converges to zero while $\|b_i - u_i \| \to \|b_i - x_0\| > 0$.
 Now observe that $$\varphi(x_0) \leq t_i \varphi(u_i) + (1-t_i)\varphi(b_i) \leq t_i (\varphi(x_0) - \epsilon) + (1-t_i)M.$$ Since $t_i \to 1$, this upper bound converges to $\varphi(x_0) - \epsilon$ which is a contradiction. \end{proof}

\begin{prop} Let $U \subset \bR^d$ be a convex set. Let $t_1, \ldots, t_n \in [0,1]$ such that $t_1 + \cdots + t_n = 1$. We have $$\sum_{i=1}^n t_i u_i \in U$$ for all $u_1, \ldots, u_n \in U$. Moreover, if $\varphi:U \to \bR$ is convex then $$\varphi \left(\sum_{i=1}^n t_i u_i \right) \leq \sum_{i=1}^n t_i \varphi(u_i). $$

\end{prop}

\begin{proof} The $n=2$ case is the definition of convexity. We proceed by induction on $n$. Write $t = t_1 + \ldots + t_{n-1}$. The result is clearly true if $t = 0$, as in that case $t_n = 1$, so we assume $t>0$. Observe that $$\sum_{i=1}^nt_iu_i = t\sum_{i=1}^{n-1}t^{-1}t_i u_i + (1-t)u_n$$ is indeed in $U$ since by the inductive hypothesis, $$\sum_{i=1}^{n-1}t^{-1}t_i u_i \in U$$ as $\sum_{i=1}^{n-1} t^{-1}t_i = 1$. Likewise by the induction hypothesis we have
\begin{align*} \varphi \left(\sum_{i=1}^n t_i u_i \right) &= \varphi \left( t\sum_{i=1}^{n-1}t^{-1}t_i u_i + (1-t)u_n \right) \\
&\leq t \varphi\left(\sum_{i=1}^{n-1}t^{-1}t_i u_i \right) + (1-t) \varphi(u_n) \\
&\leq t\sum_{i=1}^{n-1}t^{-1}t_i \varphi(u_i) + (1-t) \varphi(u_n) \\
&= \sum_{i=1}^n t_i \varphi(u_i).
\end{align*}

\end{proof}

Note that if $v:X \to \bR^d$ is a vector valued function on a probability space $(X,\mu)$ then we can define its integral $\int_X v(x) d\mu(x)$ componentwise, assuming each component is measurable and has a well defined integral (e.g, each component is positive or $L^1$). If $$\int_X \|v\| d\mu(x) < \infty$$ then we say that $v$ is $L^1$. In this case, each component is $L^1$ and the integral of $v$ makes sense.

\begin{prop}[Compact Jensen's inequality] Let $\mu$ be a probability measure supported on a compact subset $K \subset \bR^d$ and suppose $\varphi:\bR^d \to \bR$ is convex. Then $$\varphi\left(\int_K x d\mu(x)\right) \leq \int_K \varphi(x) d\mu(x). $$

\end{prop}

\begin{proof} Note that if $\mu$ is supported on a finite set $\{u_1, \ldots, u_n\}$ with $\mu(\{u_i\}) = p_i$, then by the inequality above we have that $$ \varphi\left(\int_K x d\mu(x)\right)  = \varphi\left( \sum_{i=1}^n p_i u_i \right) \leq \sum_{i=1}^n p_i\varphi(u_i) = \int_K \varphi(x) d\mu(x) .$$ Thus the result holds for finite measures. For arbitrary probability measures on $K$, we use the proposition below and the continuity of $\phi$ and $x \mapsto x$. \end{proof}

\begin{prop} Let $\mu$ be a probability measure on a compact set $[-R,R]^d \subset \bR^d$. For each positive integer $n$, let $\cC_n$ be a partition of $[-R,R]^d$ into finitely many cubes, i.e., $$[-R,R]^d = \bigcup_{C \in \cC_n} C$$ such that each $C \in \cC_n$ has diameter at most $1/n$. Let $C_n \subset [-R,R]^d$ be a collection of points consisting of exactly one element from $\cC_n$, thus we have a bijection $\iota:C_n \to \cC_n$ where $c \in \iota(c)$. Now define a probability measure on $C_n$ given by $\mu_n(\{c\}) = \mu(\iota(c))$ for $c \in C_n$. Then $$\int_{\bR^d} f d\mu_n \to \int_{\bR^d} f d\mu$$ for all continuous functions $f:[-R,R]^d \to \bR.$

\end{prop}

We now obtain immediately Jensen's inequality for integrable random variables. In this section, we fix compact sets $K_1 \subset K_2 \subset \cdots$ such that $$\bigcup_{n}K_n = \bR^d. $$

\begin{thm}[Jensen's inequality] Let $\mu$ be a probability measure  on $\bR^d$ such that $$\int_{\bR^d} \|x \| d\mu(x) < \infty$$ and suppose $\varphi:\bR^d \to \bR$ is convex. Then $$\varphi\left(\int_{\bR^d} x d\mu(x)\right) \leq \liminf_{n \to \infty} \int_{K_n} \varphi(x) d\mu(x). $$

\end{thm}

\begin{proof} For large enough $n$, we have $\mu(K_n) > 0$. By applying the Compact Jensen's inequality to the probability measure $\mu_n = \mu|_{K_n}$, which is defined by $\mu_n(A) = \mu(K_n)^{-1}\mu(A \cap K_n)$, we have 

$$\varphi\left(\mu(K_n)^{-1}\int_{K_n} x d\mu(x)\right) \leq \mu(K_n)^{-1}\int_{K_n} \varphi(x) d\mu(x). $$ Now as $n \to \infty$ we have $\mu(K_n) \to 1$ and so the continuity of $\varphi$ completes the proof. \end{proof}

Another way of stating this is as follows.

\begin{thm} Let $\nu$ be a probability measure on a space $\Omega$. Suppose that $f:\Omega \to \bR^d$ is $L^1$, i.e., $$\int_{\bR^d} \|f(\omega)\| d\nu(\omega) < \infty.$$ Suppose that $\varphi:\bR^d \to \bR$ is convex. Then 
$$\varphi \left(\int_{\Omega} f(\omega) d\nu(\omega) \right) \leq \liminf_{N \to \infty} \int_{f^{-1}(K_n)} \varphi(f(\omega)) d\nu(\omega).$$ 
In particular, if $\varphi \geq 0$ or $\varphi \circ f \in L^1(\Omega)$ then 
$$\varphi \left(\int_{\Omega} f(\omega) d\nu(\omega) \right) \leq \int_{\Omega} \varphi(f(\omega)) d\nu(\omega).$$ 
\end{thm}

\section{Haussdorf measure}

\begin{mydef} Fix $s > 0$. Let $\alpha_s>0$ be chosen such that $\alpha_s$ is the volume of the unit $s$-ball. For $A \subset \bR^d$ (not necessarily Borel) we define 
$$H^s_{\delta} (A) = \inf \left\{ \sum_{i=1}^{\infty} \alpha_s r_i^s ~|~ A \subset \bigcup_{i=1}^{\infty} B(x_i, r_i) \quad\text{for some } r_1,r_2, \ldots \in [0,\delta] \text{ and } x_1,x_2,\ldots \in \bR^d. \right\}.$$ We define the $s$-dimensional Haussdorf measure to be 
$$H^s(A) = \lim_{s \to 0} H^s_{\delta}. $$

\end{mydef}

In other words $H^s_{\delta}(A)$ measures the optimal way of covering $A$ with balls of radius at most $\delta$, where each ball of radius $r$ has measure $\alpha_s r^d$. 

It is clear the $H^s$ is an outer measure on $\bR^d$.

\begin{mydef} Let $\mu$ be an outer measure on a metric space $X$. We say that $\mu$ is a metric outer measure if it has the property that whenever $A_1,A_2 \subset X$ are such that $d(A_1, A_2) > 0$, then $\mu(A_1 \cup A_2) = \mu(A_1) + \mu(A_2).$ 

\end{mydef}

Note that the Haussdorf measure is a metric outer measure as can be seen by the following argument. If the distance between $A_1,A_2 \subset \bR^d$ is strictly greater than $\epsilon>0$, then any ball of radius at most $\epsilon$ cannot intersect both $A_1$ and $A_2$. Thus for $0 < \delta <\epsilon$ we have that 
$$H^s_{\delta}(A_1 \cup A_2) = H^s_{\delta}(A_1) + H^s_{\delta}(A_2).$$

\begin{lemma} Let $\mu$ be a metric outer measure on a metric space $X$. Suppose that $A \subset X$ is such that $\mu(A) < \infty$ and suppose that $A_1 \subset A_2 \subset \ldots$ are such that $A = \bigcup_{n \geq 1} A_n$ and $$d(A \setminus A_{n+1}, A_n) >0 $$ for all $n \geq 1$. Then $$\lim_{n \to \infty} \mu(A_n) = \mu(A).$$

\end{lemma}

\begin{proof} Let $B_n = A_{n+1}\setminus A_n$. Now observe that $B_1,B_3, \ldots, B_{2N-1}$ are all disjoint sets and the distance between any two is non-zero. Thus by definition of metric measure we have that $$\sum_{i=1}^N\mu(B_{2i-1}) = \mu(\bigcup_{i=1}^N B_{2i-1}) \leq \mu(A) < \infty.$$ The same argument holds for even indexes. Thus $$\sum_{n=1}^{\infty} \mu(B_n) < \infty.$$ 
Now $$\mu(A \setminus A_n) = \sum_{i=n+1}^{\infty} \mu(B_i) \to 0 \quad\text{as } n \to \infty.$$

Thus from $\mu(A) \leq \mu(A \setminus A_n) + \mu(A_n)$ we see that $$\mu(A) \leq \lim_{n \to \infty} \mu(A_n).$$

This completes the proof as $\mu(A_n) \leq \mu(A)$ holds. \end{proof}

\begin{thm} Let $\mu$ be a metric outer measure on a metric space $X$. Then each Borel set is $\mu$-measurable.

\end{thm}

\begin{proof} We will show that all closed sets are Borel measurable and this will be enough be since the $\mu$-measurable sets form a sigma algebra.

Thus let $C \subset X$ be closed. We must show that for all $A \subset X$ we have that $$\mu(A) \geq \mu(A \cap C) + \mu(A')$$ where $A' = A \setminus C$. This holds if $\mu(A) = \infty$, so we assume $\mu(A) < \infty$. 

Now let $$A_n = \{a \in A ~|~ d(a, C) > \frac{1}{n} \}.$$ Now that $A_n \subset A_{n+1}$ and as $C$ is closed, we have that $$\bigcup_{n=1}^{\infty} A_n = A'.$$ 

Note also that if $x \in A_n$ and $y \in A' \setminus A_{n+1}$ then we must have $$d(x,y) \geq \frac{1}{n} - \frac{1}{n+1}$$ which can be seen as follows: We have $d(y, C) \leq 1/(n+1)$ and so there exists $c \in C$ so that $$d(y,c) < \frac{1}{n+1} + \epsilon.$$ However $$\frac{1}{n} < d(x,c) \leq d(x,y) + d(y,c) \leq d(x,y) + \frac{1}{n+1} + \epsilon.$$ Thus we have $d(A_n, A' \setminus A_{n+1}) >0,$ so the previous Lemma applies. This means that $$\mu(A \cap C) + \mu(A') =\mu(A \cap C) + \lim_{n \to \infty} \mu(A_n) = \lim_{n \to \infty} \mu((A \cap C) \cup \mu(A_n)) \leq \mu(A),$$ where we use the identity $\mu((A \cap C) \cup \mu(A_n)) = \mu(A \cap C) + \mu(A_n)$ since $d(A \cap C, A_n) \geq 1/n > 0$ and $\mu$ is a metric measure. \end{proof}

Thus the Haussdorf measure forms a Borel measure (i.e., it is $\sigma$-additive on Borel sets).

\begin{prop}\label{prop: Haussdorf is Lebesgue} For (not necessarily measurable) $B \subset \bR^d$ we have $$H^d(B) = \cL_d(B),$$ i.e., the $d$-dimensional Haussdorf measure is equal to the $d$-dimensional Lebesgue measure. In fact, $H^d(B) = H^d_{\delta}(B)$ for all $\delta>0$.

\end{prop}

To prove this, we need some results about covering open sets with balls.

\begin{lemma} Let $\cC$ be a collection of (open or closed) balls in $\bR^d$ with non-zero radius. Then there exists $\cC' \subset \cC$ such that $\cC'$ is a maximal disjoint subcollection of $\cC$ (i.e., the elements of $\cC'$ are disjoint and for and $B \in \cC \setminus \cC'$ we have that $B \cap B' \neq \emptyset$ for some $B' \in \cC'$). Moreover, any such $\cC'$ must be countable.

\end{lemma}

\begin{proof} To see that any such $\cC'$ is countable, note that each element of $\cC'$ contains some rational point (point in $\bQ^d$). But there are only finitely many such rational points and no two such balls can share such a point.

To show existence of $\cC'$, we can use Zorn's lemma. But let us avoid that. We recursively select a (finite or countable) sequence of balls $B_1, B_2, \ldots \in \cC$ as follows. Let us order the rational points $q_1,q_2,\ldots \in \bQ^d$. Let $i(B) = \min\{ i ~|~ q_i \in B \}$. We recursively let $B_n$ be chosen so that $$i(B_n) = \min \{ i(B) ~|~ B \in \cC \setminus \cB_{n-1} \text{ and } B \cap B' = \emptyset \quad \text{for all } B' \in \cB_{n-1} \}$$ where $\cB_0 = \emptyset$ and $\cB_n = \{B_1, \ldots, B_n\}$. That is, we keep selecting $B_n$ so that is it disjoint from $B_1, \ldots, B_{n-1}$ and $i(B_n)$ is minimal. Note that $i(B_n)$ is strictly increasing sequence. We claim that $\cC' = \{B_1,B_2,\ldots \}$ is maximal disjoint. This is clear if $\cC'$ is finite. Thus suppose $\cC'$ is infinite and not maximal disjoint. Hence there exists $B \in \cC$ so that $B \cap B' = \emptyset$ for all $B' \in \cC'$. Now let $q_i$ be so that $i = i(B)$. As $i(B_k)$ is strictly increasing in $k$, we have that $i(B_{i+1}) > i$, which violates the choice of $B_{i+1}$. \end{proof}

If $B = B(x, r)$ is an open ball, let $D_5 B = B(x, 5r)$ denote the ball with the same centre but $5$ times the radius. 

\begin{thm}[Vitali's covering Lemma]Suppose that $\cC$ is a collection of non-empty open balls in $\bR^d$ such that $$D := \sup \{ diam(B) ~|~ B \in \cC \} < \infty$$ and let $U = \bigcup_{B \in \cC} B$. Then there exists a \textbf{disjoint} subcollection $\cC' \subset \cC$ so that $$U \subset \bigcup_{B \in \cC'} D_5B.$$ Moreover, such a $\cC'$ must be countable.

\end{thm}

\begin{proof} Let $$\cF_j = \{ B \in \cC ~|~ 2^{-j}D< diam(B) \leq 2^{-j+1} D$$ for $j = 1,2,\ldots$. Note that the $\cF_j$ partition $\cF$. Now let $\cC'_1$ be a maximal disjoint collection of $\cF_1$ and for $j > 1$ we define recursively $\cC_j$ to be a maximal disjoint collection of elements of $$\{ B \in \cF_j ~|~ B \cap B' = \emptyset \text{ for all } B' \in \bigcup_{i=1}^{j-1}\cC'_{i}\}.$$

 Now let $$\cC' = \bigcup_{j=1}^{\infty} \cC'_j.$$ Clearly the elements of $\cC'$ are disjoint. Now let $B \in \cC \setminus \cC'$. Then $B \in \cF_j$ for some $j$. As $B \notin \cC_j$ this means that either $B$ intersects non-trivially some element in $\bigcup_{i=1}^{j-1}\cC'_i$ or it intersects non-trivially some element of $\cC'_j$. In either case, it intersects non-trivially some element $B'$ of $\bigcup_{i=1}^j \cC'_i$. This means that $$diam(B') > 2^{-j}D = \frac{1}{2} 2^{-j+1}D \geq \frac{1}{2} diam(B).$$ Thus $B \subset D_5B'$. Hence as $B$ was arbitrary, this shows that $$U \subset \bigcup_{B \in \cC'} D_5B.$$  \end{proof}

Our next result shows that we can almost write any open set as a countable disjoint union of balls. Here almost means that the union of these balls is a subset of our open set and the remainder has measure zero.

\begin{thm} Let $U \subset \bR^d$ be an open set and $\delta < 0$. Then there exists a disjoint (hence countable) collection $\cC$ of open balls in $\bR^d$ such that $diam(B) < \delta$ and $B \subset U$ for each $B \in \cC$ and $$\cL_d\left(U \setminus \bigcup_{B \in \cC} B \right) = 0.$$

\end{thm}

\begin{proof} 

We reduce to the case where $U$ is bounded and hence has $\cL_d(U) < \infty$ since, after deleting a set of measure $0$, we can partition any such $U$ into bounded open sets. For example, after removing the sphere $\{ u \in U ~|~ |u| = n \}$ for each $n \in \bZ$.

Let $\cC_0$ be the collection of non-empty open balls that are subsets of $U$ and have diameter at most $\delta$. Applying the Vitali covering lemma to this, we can choose a subset $\cC_1 \subset \cC_0$ of disjoint balls so that $$V_1 = \bigcup_{B \in \cC_1} D_5 B \supset \bigcup_{B \in \cC_0}B = U. $$ Let $$W_1 = \bigcup_{B \in \cC_1} B.$$

Thus $$\cL_d(U) \leq \cL_d(V_1) \leq \sum_{B \in \cC_1} 5^d\cL_d(B) = 5^d \cL_d(W_1).$$ This means that if we let $\alpha = 5^{-d}$ then $$\cL_d(W_1) \geq \alpha \cL_d(U).$$ Now fix $0< \beta < \alpha$. This means that we can choose a finite subset $\cF_1 \subset \cC_1$ such that $\cL_d(F_1) \geq \beta \cL_d(U)$ where $$F_1 = \bigcup_{B \in \cF_1} \overline{B}.$$

Now let $U_1 = U \setminus F_1$ and note that $U_1$ is open (as $F_1$, being the union of finitely many closed balls is closed). However $$\cL_d(U_1) = \cL_d(U) - \cL_d(F_1) \leq (1 - \beta) \cL_d(U).$$  

Thus we can repeat the same procedure to $U_1$ in place of $U$ and find a finite set $\cF_2$ of open balls in $U_1$ of diameter at most $\delta$ such that if we delete the closure of their union then we get a set $U_2 \subset U_1$ of measure $$\cL_d(U_2) \leq (1 - \beta) \cL_d(U_1) \leq (1-\beta)^2 \cL_d(U).$$ Repeating this procedure, we can finally let $$\cC = \bigcup_{n=1}^{\infty} \cF_n.$$ Note that this is a disjoint collection by construction. The complement of the union of all $\cC$ has measure $0$ since $$\cL_d(U_n) \leq \beta ^n \cL_d(U) \to 0 \quad\text{as } n \to \infty.$$
\end{proof}

\begin{proof}[Proof of Proposition~\ref{prop: Haussdorf is Lebesgue} ] We first show the easy inequality $$\cL_d(B) \leq H^d(B).$$ This follows since $\alpha_d r^d$ is the Lebesgue measure of a ball of radius $r$ and thus $$\cL_d(B) \leq \sum_{i=1}^{\infty} \alpha_d r_i^d$$ for any cover $\bigcup_{i=1}^{\infty} B(x_i, r_i)$ of $B$. Thus $\cL_d(B) \leq H^d_{\delta}(B)$ holds for all $\delta>0$.

Now we prove the reverse inequality. It is sufficient to prove it when $B = U$ is an open set (i.e., if $\mu$ is $H^d_{\delta}$ or Lebesgue, then there is always an open set $V$ containing $B$ so that $\mu(B) \leq \mu(V) < \mu(B) + \epsilon$, thus $\mu(B)$ is the infinum of $\mu(V)$ over all such $V$).

For this, we know by the previous Theorem that we can write $$U = Q \cup \bigcup_{i=1}^{\infty} B_i$$ where each $B_i$ is an open ball of radius at most $\delta$ and $\cL_d(Q) = 0$. It follows that $H^d_{\delta}(Q) = 0$ since $Q$ can be covered with cubes of very small volume and thus balls of volume just a constant multiple of that, hence also arbitrarily small. In particular, this means that if we fix $\epsilon>0$ then we can cover $Q$ with balls $B'_1,B'_2, \ldots$ of radius at most $\delta$ such that $$\sum_{i=1}^{\infty} \cL_d(B'_i) < \epsilon.$$ Thus $$H^d_{\delta}(U) \leq \sum_{i=1}^{\infty} \cL_d(B_i) + \sum_{i=1}^{\infty} \cL_d(B'_i) = \cL_d(U) + \sum_{i=1}^{\infty} \cL_d(B'_i)  \leq \cL_d(U) + \epsilon. $$ Taking $\epsilon \to \infty$ completes the proof.

\end{proof}

\section{Area formula}

Suppose $X \subset \bR^n$ is open and $f:X \to \bR^n$ with $n \leq m$ is coninuously differentiable. We let $Df(x)$ denote the derivative of $f$ at $x \in X$, which is a linear map $Df(x):\bR^n \to \bR^m$ where $$Df(x)u = \lim_{t \to 0} t^{-1}(f(x+tu) - u).$$ Recall that to say that it is continously differentiable is to say that $x \mapsto Df(x)$ is a continuous map in the operator norm on the space of linear maps $\bR^n \to \bR^m$. 
We define the Jacobian to be $$Jf(x) = \sqrt{\det(Df(x)^*Df(x))}$$ where $Df(x)^*:\bR^m \to \bR^n$ denotes the transpose (adjoint) of $Df(x):\bR^n \to \bR^m$.

The goal of this section is to prove the following.

\begin{thm}[Area formula] Suppose that $f:X \to \bR^m$, where $X \subset \bR^n$ is open and $n \leq m$, is continuously differentiable on $X$ and is injective. Then for Borel $B \subset X$ we have that $$H^n(f(B)) = \int_B Jf(x) d\cL_n(x).$$

\end{thm}

\begin{eg}[Arc-length] Suppose $n=1$. Thus $f(x) = [f_1(x), \ldots, f_n(x)]^t$ is a column vector. Thus $Df(x)$ is the column vector of derivatives $[f'_1(x), \ldots, \partial_n f'(x)]^t$. Thus $$Jf(x) = \sqrt{\sum_{i=1}^m (f'_1(x))^2} = \| Df(x)\|$$ is magnitude of the velocity vector (if $x$ denotes time). Thus $$ H^1(f([a,b])) = \int_{a}^b \|Df(x)\| dx$$ is the arc-length of the parametrized curve $\{ f(x) ~|~ x \in [a,b]\}$.

\end{eg}

We fix for the rest of this section $f$, $X$ and $n,m$ as in the Area formula.

\subsection{Area formula for Linear maps} We first examine the linear case where $f(x) = Lx$ is a linear map. Thus $L:\bR^n \to \bR^m$ has trivial kernel. By the Singular Value Decomposition, we may write $L = UDV$ where $U:\bR^n \to \bR^m$ and $V:\bR^n \to \bR^n$ are isometries and $D:\bR^n \to \bR^n$ is a diagonal matrix with positive entries on the diagonal.

By Proposition~\ref{prop: Haussdorf is Lebesgue} and the fact that the Haussdorf measure is rotation invariant, we have $$\cL_n(VB) = H^n(VB) = H^n(B) = \cL_n(B)$$ for any Borel $B \subset \bR^n$. Thus $\cL_n(DVB) = \det(D) \cL_n(VB) = \cL_n(B)$ as $D$ is diagonal (i.e., the image of each rectangle is another rectangle where the sidelengths get scaled by the diagonal elements of $D$).

\begin{lemma} Let $U:\bR^n \to \bR^m$ be an isometry (as above), then for borel $B \subset \bR^n$ we have that $H^n(UB) = \cL_n(B)$.

\end{lemma}

\begin{proof} Note that since $H^n$ is rotation invariant, we may assume that $U$ is the standard embedding of $\bR^n$ into $\bR^m$ (i.e., $(x_1, \ldots, x_n) \mapsto (x_1, \ldots, x_n, 0, \ldots, 0)$). It thus suffices to show that $\cL_n(B) = H^n(B \times \{0\}^{m-n})$. This follows since for any covering of $B \times \{0\}^{n-m}$ by balls in $\bR^m$, if we project each ball onto $\bR^n$ then we get a ball of the same radius. \end{proof}

We thus get that $H^n(LB) = \det(D)\cL_n(B)$, i.e., $$H^n(f(B)) = \int_B \det(D) d\cL_n(x).$$
